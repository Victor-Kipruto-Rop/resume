Quickstart Example (Node.js)

POST /v1/transactions/verify 
const PesaGuard = require('pesaguard');
const pguard = new PesaGuard({ apiKey: 'pg_live_98f7d6c5a4b3...' });

async function verifyTx() {
  const assessment = await pguard.transactions.verify({
    amount: 45000.00,
    currency: 'KES',
    source_account: 'ACC-8839201',
    destination_account: 'ACC-9921034'
  });
  console.log(assessment.status); // 'APPROVED' | 'FLAGGED'
}
verifyTx();
